function addContextualGreeting() {
    const now = new Date();
    const hour = now.getHours();
    const season = getCurrentSeason();
    
    let greeting;
    if (hour < 6) greeting = `🌙 Peace in the ${season} night`;
    else if (hour < 12) greeting = `🌅 ${season} morning blessings`;
    else if (hour < 18) greeting = `☀️ ${season} afternoon warmth`;
    else greeting = `🌟 ${season} evening gratitude`;
    
    // Add to hero section
    const heroContent = document.querySelector('.hero-content');
    if (heroContent) {
        const greetingElement = document.createElement('p');
        greetingElement.textContent = greeting;
        greetingElement.style.fontSize = '1.2rem';
        greetingElement.style.marginBottom = '1rem';
        greetingElement.style.opacity = '0.9';
        
        heroContent.insertBefore(greetingElement, heroContent.querySelector('h1'));
    }
}

function getCurrentSeason() {
    const month = new Date().getMonth() + 1; // 1-12
    
    if (month >= 3 && month <= 5) return 'spring';
    if (month >= 6 && month <= 8) return 'summer'; 
    if (month >= 9 && month <= 11) return 'autumn';
    return 'winter';
}

function setupCommunityFeatures() {
    // Add gentle interactions that build connection
    const valueCards = document.querySelectorAll('.value-card');
    
    valueCards.forEach(card => {
        card.addEventListener('click', function() {
            // Gentle feedback when exploring values
            const value = this.querySelector('h3').textContent;
            console.log(`💖 Exploring: ${value} - Thank you for caring about our values!`);
            
            // Subtle visual feedback
            this.style.background = 'var(--light-sage)';
            setTimeout(() => {
                this.style.background = 'white';
            }, 1000);
        });
    });
}

function trackCommunityWellbeing() {
    // Privacy-preserving analytics: Track what helps, not who visits
    const wellbeingMetrics = {
        page_helpfulness: 0,
        community_connections: 0,
        learning_engagements: 0
    };
    
    // Only track interactions that improve community service
    document.addEventListener('click', function(e) {
        if (e.target.matches('.helpful-content')) {
            wellbeingMetrics.page_helpfulness++;
        }
        if (e.target.matches('.community-connect')) {
            wellbeingMetrics.community_connections++;
        }
        if (e.target.matches('.learning-resource')) {
            wellbeingMetrics.learning_engagements++;
        }
        
        // Store locally, never send to external servers
        localStorage.setItem('community_wellbeing', JSON.stringify(wellbeingMetrics));
    });
    
    console.log('🛡️ Privacy-preserving community insights enabled');
}

                if (entry.isIntersecting) {
                    this.loadEnhancedFeature(entry.target);
                }
            });

                break;

### **Challenge: Loading Speed** ⚡📊
**Guardian Dragon approach**: Respect for user resources

```javascript
// Lazy load images with intersection observer
function setupResponsibleImageLoading() {
    const images = document.querySelectorAll('img[data-src]');
    
    const imageObserver = new IntersectionObserver((entries, observer) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                const img = entry.target;
                img.src = img.dataset.src;
                img.classList.remove('lazy');
                imageObserver.unobserve(img);
                
                console.log(`🖼️ Image loaded with respect for bandwidth`);
            }
        });
    });
    
    images.forEach(img => imageObserver.observe(img));
}
```

### **Challenge: User Privacy** 🔐👥
**Guardian Dragon approach**: Privacy by design

```javascript
// No external tracking, only local functionality
class PrivacyPreservingAnalytics {
    constructor() {
        this.localMetrics = JSON.parse(localStorage.getItem('site_metrics') || '{}');
    }
    
    trackPageHelpfulness(page, helpful) {
        // Only track if user explicitly indicates helpfulness
        if (!this.localMetrics[page]) {
            this.localMetrics[page] = { helpful: 0, total: 0 };
        }
        
        this.localMetrics[page].total++;
        if (helpful) this.localMetrics[page].helpful++;
        
        localStorage.setItem('site_metrics', JSON.stringify(this.localMetrics));
        
        console.log('📊 Feedback recorded locally - your privacy preserved');
    }
    
    // Never send data to external servers
    // Never track individual behavior
    // Never store personal information
}
```
